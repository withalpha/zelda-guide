=====================================================
如果 GitHub 访问不了，用这个国内镜像版本
复制下面所有命令，粘贴到路由器 SSH 终端中执行
=====================================================

# ===== 使用国内镜像加速 =====
echo "正在检测设备架构..."
ARCH=$(opkg print-architecture | grep -v "all\|noarch" | awk '{print $2}' | head -n 1)
echo "设备架构: $ARCH"

# 更新软件源
opkg update

# 备份配置
cp /etc/opkg/customfeeds.conf /etc/opkg/customfeeds.conf.bak 2>/dev/null

# 添加镜像源
cat >> /etc/opkg/customfeeds.conf << 'EOF'

# Passwall2 依赖源（镜像加速）
src/gz passwall_packages https://mirror.ghproxy.com/https://github.com/xiaorouji/openwrt-passwall-packages/releases/download/latest
EOF

opkg update

# 安装依赖
echo "安装依赖包..."
opkg install coreutils coreutils-base64 coreutils-nohup lyaml tcping geoview v2ray-geoip v2ray-geosite

# 进入临时目录
cd /tmp

# 使用镜像下载（如果文件已存在则跳过）
if [ ! -f "luci-app-passwall2_26.6.3-r1_all.ipk" ]; then
    echo "下载 Passwall2 主程序（使用镜像）..."
    wget --no-check-certificate https://mirror.ghproxy.com/https://github.com/Openwrt-Passwall/openwrt-passwall2/releases/download/26.6.3-1/luci-app-passwall2_26.6.3-r1_all.ipk
fi

if [ ! -f "luci-i18n-passwall2-zh-cn_26.6.3_all.ipk" ]; then
    echo "下载中文语言包（使用镜像）..."
    wget --no-check-certificate https://mirror.ghproxy.com/https://github.com/Openwrt-Passwall/openwrt-passwall2/releases/download/26.6.3-1/luci-i18n-passwall2-zh-cn_26.6.3_all.ipk
fi

# 安装
echo "安装 Passwall2..."
opkg install luci-app-passwall2_26.6.3-r1_all.ipk
opkg install luci-i18n-passwall2-zh-cn_26.6.3_all.ipk

# 重启服务
/etc/init.d/uhttpd restart

echo ""
echo "✅ 安装完成！"
echo "请刷新路由器管理页面"
