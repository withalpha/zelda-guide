# GL-MT3000 路由器配置命令
# 使用PuTTY或其他SSH工具连接到路由器

# ============================================
# 步骤1: SSH连接路由器
# ============================================
# 地址: 192.168.8.1
# 用户名: root
# 密码: 你的路由器密码

# ============================================
# 步骤2: 检查PassWall和Xray版本
# ============================================
opkg list-installed | grep passwall
opkg list-installed | grep xray

# ============================================
# 步骤3: 更新Xray核心（支持XHTTP）
# ============================================
opkg update
opkg upgrade xray-core
/etc/init.d/passwall restart

# ============================================
# 步骤4: 检查是否支持XHTTP
# ============================================
xray version

# 如果版本 >= 1.8.0，应该支持XHTTP

# ============================================
# 步骤5: 查看PassWall配置文件位置
# ============================================
ls -la /etc/config/passwall
ls -la /usr/share/passwall/

# ============================================
# 手动方式：直接在Web界面配置
# ============================================
# 1. 浏览器访问: http://192.168.8.1
# 2. 进入：服务 → PassWall
# 3. 点击：节点列表 → 添加
# 4. 按照之前给你的参数填写

# ============================================
# 如果需要安装OpenClash（替代方案）
# ============================================
opkg update
opkg install luci-app-openclash

# 然后将 openclash-config.yaml 上传到路由器
